💡 Minimal Quick-Start Examples for Developers
🅰️ Vanilla JavaScript (Pure HTML + JS)
💾 Download JS Script (dh-grid.js) 
<!DOCTYPE html>
<html>
<head>
  <script src="resources/dadhawk/js/dh-grid.js"></script>
</head>
<body>
  <dh-grid-element id="myGrid" rows="5" cols="5"></dh-grid-element>

  <script>
    const grid = document.getElementById('myGrid');
    // Composite Tree Captions & Custom Components
    grid.setAttribute('captions', JSON.stringify([
      "Sales / H1 / Q1", "Sales / H1 / Q2",
      "Sales / H2 / Q3", "Sales / H2 / Q4", "Status"
    ]));
    grid.setAttribute('content', JSON.stringify([
      ["Header", "Q1", "Q2", "Q3", "Status"],
      ["2026", "$120k", "$85k", "29.1%", "Completed"]
    ]));
    grid.setAttribute('components', JSON.stringify({
      "r1_c4": "status-selector"
    }));
    grid.addEventListener('cell-change', (e) => {
      console.log(e.detail.row, e.detail.col, e.detail.value);
    });
  </script>
</body>
</html>
🅱️ Minimal Quick-Start Examples for JSF Developers (JSF View + Bean)
📦 Download Maven JAR (v1.0.0-RC1) 
<!-- page.xhtml -->
<html xmlns="http://www.w3.org/1999/xhtml"
      xmlns:h="jakarta.faces.html"
      xmlns:f="jakarta.faces.core"
      xmlns:dh="http://dadhawk.com/faces">
<h:head><title>JSF Grid</title></h:head>
<h:body>
  <h:form id="gridForm">
    <dh:dhGrid id="myGrid"
               rowCount="#{gridBean.rowCount}"
               colCount="#{gridBean.colCount}"
               content="#{gridBean.content}"
               captions="#{gridBean.captions}"
               componentMap="#{gridBean.componentMap}" />

    <!-- Save Grid to Backend Managed Bean -->
    <h:commandButton value="💾 Save Grid to Backend" action="#{gridBean.saveGrid()}">
      <f:ajax execute="@form" render="saveMsg" />
    </h:commandButton>
    <h:outputText id="saveMsg" value="#{gridBean.lastSaveStatus}" />
  </h:form>
</h:body>
</html>

// Java CDI Backing Bean (@Named @SessionScoped):
@Named("gridBean")
@SessionScoped
public class GridBean implements Serializable {
    private String[][] content;
    private String lastSaveStatus;

    public String saveGrid() {
        // Save action method automatically receives updated matrix content
        this.lastSaveStatus = "Saved grid matrix to GridBean!";
        return null;
    }
}
🚀 Downloads & Maven Repository Integration
1. CLI Command to Install JAR to Local Repository (~/.m2): 
mvn install:install-file \
  -Dfile=dhgrid-component-1.0.0-RC1.jar \
  -DgroupId=com.dadhawk.faces \
  -DartifactId=dhgrid-component \
  -Dversion=1.0.0-RC1 \
  -Dpackaging=jar
2. Add Maven Dependency to your pom.xml: 
<dependency>
    <groupId>com.dadhawk.faces</groupId>
    <artifactId>dhgrid-component</artifactId>
    <version>1.0.0-RC1</version>
</dependency>
